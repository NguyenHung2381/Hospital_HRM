usp_Recalc_All|SQL_STORED_PROCEDURE|NULL|
CREATE PROCEDURE [dbo].[usp_Recalc_All]
  @id_report INT = NULL
-- NULL = tính lại tất cả báo cáo
AS
BEGIN
  SET NOCOUNT ON;

  -- Cập nhật cả recommended_staff (TT03) và recommended_staff_calc (Khuyến nghị)
  UPDATE r
    SET
        r.[recommended_staff]      = CAST(v.tt03_staff_calc    AS SMALLINT),
        r.[recommended_staff_calc] = CAST(v.rec_staff_calc     AS SMALLINT),
        r.[updated_at]             = SYSDATETIMEOFFSET()
    FROM [dbo].[Report_Department_Records ] r
    INNER JOIN [dbo].[vw_TT03_Recommended] v
    ON r.[Id] = v.record_id
    WHERE
        (v.tt03_staff_calc IS NOT NULL OR v.rec_staff_calc IS NOT NULL)
    AND (@id_report IS NULL OR r.[id_report] = @id_report);

  PRINT CAST(@@ROWCOUNT AS NVARCHAR) + N' bản ghi đã được cập nhật (TT03 + Khuyến nghị).';
END

vw_TT03_Recommended|VIEW|NULL|
CREATE VIEW [dbo].[vw_TT03_Recommended]
AS
  SELECT
    r.[Id]                          AS record_id,
    r.[id_report],
    r.[id_department],
    d.[name_department],
    d.[code_department],

    -- ── Công thức TT03 ─────────────────────────────────────
    cfg_tt.[formula_type]           AS tt03_formula_type,
    cfg_tt.[note]                   AS tt03_formula_note,

    -- Đầu vào
    r.[patient_level_1]             AS L1,
    r.[patient_level_2]             AS L2,
    r.[patient_level_3]             AS L3,
    ISNULL(r.[patient_level_1],0)
      + ISNULL(r.[patient_level_2],0)
      + ISNULL(r.[patient_level_3],0) AS total_NB,
    r.[outpatient_cnt],
    d.[bed_count],

    -- Nhân lực hiện tại
    r.[total_staff],
    r.[staff_working],

    -- TT03 đã lưu (giữ nguyên)
    r.[recommended_staff]           AS tt03_staff_saved,

    -- TT03 tính lại theo công thức
    CAST(
        CASE cfg_tt.[formula_type]
            WHEN 'standard' THEN
                ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio]
                  / cfg_tt.[shift_divisor]
                  * cfg_tt.[shift_multiplier]
                ) + ISNULL(cfg_tt.[fixed_add], 0)
            WHEN 'icu' THEN
                ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio]
                  / cfg_tt.[shift_divisor]
                  * cfg_tt.[shift_multiplier]
                )
            WHEN 'surgery' THEN
                ( CAST( ISNULL(d.[bed_count], 0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio]
                  / cfg_tt.[shift_divisor]
                  * cfg_tt.[shift_multiplier]
                )
            WHEN 'custom_coef' THEN
                ( ISNULL(r.[patient_level_1],0) * CAST(d.[coef_level_1] AS DECIMAL(10,3)) )
              + ( ISNULL(r.[patient_level_2],0) * CAST(d.[coef_level_2] AS DECIMAL(10,3)) )
              + ( ISNULL(r.[patient_level_3],0) * CAST(d.[coef_level_3] AS DECIMAL(10,3)) )
              + ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * CAST(d.[coef_total] AS DECIMAL(10,3)) )
            ELSE NULL
        END
    AS DECIMAL(8,2))                AS tt03_staff_calc,

    -- ── Nhân lực Khuyến nghị ───────────────────────────────
    cfg_rc.[formula_type]           AS rec_formula_type,
    cfg_rc.[note]                   AS rec_formula_note,

    -- Khuyến nghị đã lưu
    r.[recommended_staff_calc]      AS rec_staff_saved,

    -- Khuyến nghị tính lại
    CAST(
        CASE cfg_rc.[formula_type]

            -- Phổ biến: L1*coef_l1 + L2*coef_l2 + L3*coef_l3 + fixed_add
            WHEN 'coef' THEN
                ( ISNULL(r.[patient_level_1],0) * CAST(cfg_rc.[coef_l1] AS DECIMAL(10,4)) )
              + ( ISNULL(r.[patient_level_2],0) * CAST(cfg_rc.[coef_l2] AS DECIMAL(10,4)) )
              + ( ISNULL(r.[patient_level_3],0) * CAST(cfg_rc.[coef_l3] AS DECIMAL(10,4)) )
              + ISNULL(cfg_rc.[fixed_add], 0)

            -- Cấp cứu: (L1+L2+L3) * ratio
            WHEN 'total_ratio' THEN
                CAST( ISNULL(r.[patient_level_1],0)
                    + ISNULL(r.[patient_level_2],0)
                    + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                * ISNULL(cfg_rc.[outpatient_ratio], 0)
              + ISNULL(cfg_rc.[fixed_add], 0)

            -- Gây mê: outpatient_cnt * ratio
            WHEN 'outpatient_count' THEN
                CAST( ISNULL(r.[outpatient_cnt], 0) AS DECIMAL(10,3) )
                * ISNULL(cfg_rc.[outpatient_ratio], 0)
              + ISNULL(cfg_rc.[fixed_add], 0)

            ELSE NULL
        END
    AS DECIMAL(8,2))                AS rec_staff_calc,

    -- Chênh lệch TT03
    r.[staff_working] - CAST(
        CASE cfg_tt.[formula_type]
            WHEN 'standard' THEN
                ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio] / cfg_tt.[shift_divisor] * cfg_tt.[shift_multiplier]
                ) + ISNULL(cfg_tt.[fixed_add], 0)
            WHEN 'icu' THEN
                ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio] / cfg_tt.[shift_divisor] * cfg_tt.[shift_multiplier] )
            WHEN 'surgery' THEN
                ( CAST( ISNULL(d.[bed_count],0) AS DECIMAL(10,3) )
                  * cfg_tt.[patient_ratio] / cfg_tt.[shift_divisor] * cfg_tt.[shift_multiplier] )
            WHEN 'custom_coef' THEN
                ( ISNULL(r.[patient_level_1],0) * CAST(d.[coef_level_1] AS DECIMAL(10,3)) )
              + ( ISNULL(r.[patient_level_2],0) * CAST(d.[coef_level_2] AS DECIMAL(10,3)) )
              + ( ISNULL(r.[patient_level_3],0) * CAST(d.[coef_level_3] AS DECIMAL(10,3)) )
              + ( CAST( ISNULL(r.[patient_level_1],0)
                      + ISNULL(r.[patient_level_2],0)
                      + ISNULL(r.[patient_level_3],0) AS DECIMAL(10,3) )
                  * CAST(d.[coef_total] AS DECIMAL(10,3)) )
            ELSE NULL
        END
    AS DECIMAL(8,2))                AS tt03_surplus_deficit,

    r.[coordination],
    r.[note]                        AS record_note,
    r.[created_at],
    r.[updated_at]

  FROM [dbo].[Report_Department_Records ] r
    INNER JOIN [dbo].[Departments] d
    ON r.[id_department] = d.[id_department]
    LEFT JOIN [dbo].[Dept_TT03_Config] cfg_tt
    ON r.[id_department] = cfg_tt.[id_department]
    LEFT JOIN [dbo].[Dept_Recommended_Config] cfg_rc
    ON r.[id_department] = cfg_rc.[id_department];
